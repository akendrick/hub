<script>
const LAT = 49.912;
const LON = -116.910;
const WU_API_KEY = "b6a053665de6468ca053665de6068c2e";

const proxy = url => "ical-proxy.php?url=" + encodeURIComponent(url);

function updateCurrent(data) {
  document.getElementById("temp").textContent =
    Math.round(data.temperature) + "°C";
  document.getElementById("wind").textContent =
    "Wind: " + Math.round(data.windSpeed) + " km/h";
  document.getElementById("humidity").textContent =
    "Humidity: " + data.relativeHumidity + "%";
  document.getElementById("condition").textContent =
    data.wxPhraseLong || "Conditions unavailable";
}

function fetchWU() {
  const url =
    `https://api.weather.com/v3/wx/conditions/current?` +
    `apiKey=${WU_API_KEY}&format=json&language=en-CA&` +
    `geocode=${LAT},${LON}&units=m`;

  fetch(proxy(url))
    .then(r => r.json())
    .then(updateCurrent)
    .catch(fetchOpenMeteo);
}

function fetchOpenMeteo() {
  const url =
    `https://api.open-meteo.com/v1/forecast?` +
    `latitude=${LAT}&longitude=${LON}&current_weather=true`;

  fetch(proxy(url))
    .then(r => r.json())
    .then(data => {
      updateCurrent({
        temperature: data.current_weather.temperature,
        windSpeed: data.current_weather.windspeed,
        relativeHumidity: "--",
        wxPhraseLong: "Open-Meteo fallback"
      });
    })
    .catch(() => {
      document.getElementById("condition").textContent =
        "Weather unavailable";
    });
}

function fetchForecast() {
  const url =
    `https://api.open-meteo.com/v1/forecast?` +
    `latitude=${LAT}&longitude=${LON}&daily=temperature_2m_max,temperature_2m_min&timezone=auto`;

  fetch(proxy(url))
    .then(r => r.json())
    .then(data => {
      const container = document.getElementById("forecast");
      container.innerHTML = "";
      data.daily.time.slice(0,5).forEach((date, i) => {
        const div = document.createElement("div");
        div.textContent =
          date.slice(5) + "  " +
          Math.round(data.daily.temperature_2m_max[i]) + "° / " +
          Math.round(data.daily.temperature_2m_min[i]) + "°";
        container.appendChild(div);
      });
    })
    .catch(() => {
      document.getElementById("forecast").textContent =
        "Forecast unavailable";
    });
}

function refresh() {
  fetchWU();
  fetchForecast();
}

refresh();
setInterval(refresh, 600000); // 10 minutes
</script>
